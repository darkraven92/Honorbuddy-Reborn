cmake_minimum_required(VERSION 3.20)
project(hb_navigation LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(DETOUR_ROOT "" CACHE PATH "VMaNGOS dep/recastnavigation/Detour directory (64-bit refs)")
if(NOT EXISTS "${DETOUR_ROOT}/Include/DetourNavMesh.h")
  message(FATAL_ERROR "Set DETOUR_ROOT to the VMaNGOS Detour source directory")
endif()
file(GLOB DETOUR_SOURCES CONFIGURE_DEPENDS "${DETOUR_ROOT}/Source/*.cpp")
add_library(hb_navigation SHARED bridge.cpp ${DETOUR_SOURCES})
target_include_directories(hb_navigation PRIVATE "${DETOUR_ROOT}/Include")
target_compile_options(hb_navigation PRIVATE -Wall -Wextra)
enable_testing()
add_executable(navigation_native_test tests.cpp)
target_include_directories(navigation_native_test PRIVATE "${DETOUR_ROOT}/Include")
target_link_libraries(navigation_native_test PRIVATE hb_navigation)
add_test(NAME navigation_native COMMAND navigation_native_test)
